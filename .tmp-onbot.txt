package org.firstinspires.ftc.teamcode;

import com.qualcomm.hardware.rev.RevHubOrientationOnRobot;
import com.qualcomm.hardware.gobilda.GoBildaPinpointDriver;
import com.qualcomm.robotcore.eventloop.opmode.OpMode;
import com.qualcomm.robotcore.eventloop.opmode.TeleOp;
import com.qualcomm.robotcore.hardware.DcMotor;
import com.qualcomm.robotcore.hardware.DcMotorEx;
import com.qualcomm.robotcore.hardware.IMU;
import com.qualcomm.robotcore.hardware.Servo;

import org.firstinspires.ftc.robotcore.external.navigation.AngleUnit;
import org.firstinspires.ftc.robotcore.external.navigation.DistanceUnit;
import org.firstinspires.ftc.robotcore.external.navigation.Pose2D;

@TeleOp(name = "driveTurretAimTest", group = "Test")
public class turret extends OpMode {

    /* ==================== 硬件名称 ==================== */
    private static final String FRONT_LEFT_NAME  = "leftFrontMotor";
    private static final String BACK_RIGHT_NAME  = "rightBackMotor";
    private static final String BACK_LEFT_NAME   = "leftBackMotor";
    private static final String FRONT_RIGHT_NAME = "rightFrontMotor";

    private static final String INTAKE_NAME  = "intakeMotor";
    private static final String TRANSIT_NAME = "transitMotor";

    private static final String TURRET_LEFT_NAME  = "leftServo";
    private static final String TURRET_RIGHT_NAME = "rightServo";

    private static final String PINPOINT_NAME = "od";

    /* ==================== 常量 ==================== */
    private static final double STICK_DEADZONE    = 0.05;//遥感死区
    private static final double TRIGGER_THRESHOLD = 0.05;//trigger判断界限

    private static final double TURRET_INIT = 0.5;//初始化角度
    private static final double TURRET_MIN  = 0.0;
    private static final double TURRET_MAX  = 1.0;
    private static final double TURRET_SERVO_RANGE_DEG = 300.0;//舵机最大角度

    private static final double TURRET_MANUAL_STEP_DEG = 2.0;//左右按键微调角度
    private static final double TURRET_DEADBAND_DEG = 0.5;//

    private static final double TURRET_MECH_RATIO = 0.98;//传动系数
    private static final double TURRET_SIGN = 1.0;//正转+1.0/反转-1.0

    // 云台机械安装反向 180°
    private static final double TURRET_MOUNT_OFFSET_DEG = 180.0;//棍母

    /* ==================== Pinpoint ==================== */
    private static final double PINPOINT_X_OFFSET_MM = -20;
    private static final double PINPOINT_Y_OFFSET_MM = -135.2;

    private static final double START_X_IN = 72.0;//pinpo初始坐标
    private static final double START_Y_IN = 72.0;
    private static final double START_HEADING_DEG = 0.0;//朝向

    /* ==================== 双瞄点 ==================== */
    // 固定 x = 60
    private static final double AIM_POINT_X_IN = 60.0;

    // y > 72 瞄 (60, 84)；y < 72 瞄 (60, 60)
    private static final double AIM_POINT_Y_HIGH_IN = 84.0;
    private static final double AIM_POINT_Y_LOW_IN  = 60.0;
    private static final double AIM_SWITCH_Y_IN     = 72.0;

    // 切换滞回，避免在 72 附近抖动
    private static final double AIM_SWITCH_HYST_IN  = 1.0;//死区

    private static final double AIM_MIN_DIST_IN = 0.5;

    /* ==================== 硬件 ==================== */
    private DcMotorEx frontLeft, backRight, backLeft, frontRight;
    private DcMotorEx intakeMotor, transitMotor;
    private Servo turretLeft, turretRight;
    private IMU imu;
    private GoBildaPinpointDriver pinpoint;

    /* ==================== 状态 ==================== */
    private double turretPos = TURRET_INIT;
    private double turretTargetYawDeg = START_HEADING_DEG;
    private double aimOffsetDeg = 0.0;

    private boolean prevLeft, prevRight, prevStart, prevX;

    private boolean pinpointReady = false;
    private boolean needSetStartPos = true;
    private double pinX = 0.0, pinY = 0.0, pinHeadingDeg = 0.0;

    private boolean aimMode = false;
    private double aimDistance = 0.0;
    private double aimTargetDeg = 0.0;

    // 当前自瞄目标
    private double aimTargetX = AIM_POINT_X_IN;
    private double aimTargetY = AIM_POINT_Y_LOW_IN;

    private boolean hasValidAim = false;
    private double lastValidAimTargetDeg = 0.0;

    // 云台是否可达
    private boolean turretReachable = true;

    @Override
    public void init() {
        frontLeft  = hardwareMap.get(DcMotorEx.class, FRONT_LEFT_NAME);
        backRight  = hardwareMap.get(DcMotorEx.class, BACK_RIGHT_NAME);
        backLeft   = hardwareMap.get(DcMotorEx.class, BACK_LEFT_NAME);
        frontRight = hardwareMap.get(DcMotorEx.class, FRONT_RIGHT_NAME);

        intakeMotor  = hardwareMap.get(DcMotorEx.class, INTAKE_NAME);
        transitMotor = hardwareMap.get(DcMotorEx.class, TRANSIT_NAME);

        turretLeft  = hardwareMap.get(Servo.class, TURRET_LEFT_NAME);
        turretRight = hardwareMap.get(Servo.class, TURRET_RIGHT_NAME);

        frontLeft.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);
        backRight.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);
        backLeft.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);
        frontRight.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);

        frontLeft.setZeroPowerBehavior(DcMotor.ZeroPowerBehavior.BRAKE);
        backRight.setZeroPowerBehavior(DcMotor.ZeroPowerBehavior.BRAKE);
        backLeft.setZeroPowerBehavior(DcMotor.ZeroPowerBehavior.BRAKE);
        frontRight.setZeroPowerBehavior(DcMotor.ZeroPowerBehavior.BRAKE);

        frontLeft.setDirection(DcMotor.Direction.REVERSE);
        backLeft.setDirection(DcMotor.Direction.REVERSE);
        frontRight.setDirection(DcMotor.Direction.FORWARD);
        backRight.setDirection(DcMotor.Direction.FORWARD);

        intakeMotor.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);
        transitMotor.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);
        intakeMotor.setZeroPowerBehavior(DcMotor.ZeroPowerBehavior.BRAKE);
        transitMotor.setZeroPowerBehavior(DcMotor.ZeroPowerBehavior.BRAKE);

        imu = hardwareMap.get(IMU.class, "imu");
        RevHubOrientationOnRobot orientation = new RevHubOrientationOnRobot(
                RevHubOrientationOnRobot.LogoFacingDirection.RIGHT,
                RevHubOrientationOnRobot.UsbFacingDirection.UP);
        imu.initialize(new IMU.Parameters(orientation));
        imu.resetYaw();

        pinpointReady = false;
        needSetStartPos = true;
        try {
            pinpoint = hardwareMap.get(GoBildaPinpointDriver.class, PINPOINT_NAME);
            pinpoint.setOffsets(PINPOINT_X_OFFSET_MM, PINPOINT_Y_OFFSET_MM, DistanceUnit.MM);
            pinpoint.resetPosAndIMU();
            pinpointReady = true;
        } catch (Exception e) {
            pinpointReady = false;
        }

        turretPos = TURRET_INIT;
        turretTargetYawDeg = START_HEADING_DEG;
        aimOffsetDeg = 0.0;
        turretLeft.setPosition(turretPos);
        turretRight.setPosition(turretPos);

        aimMode = false;
        aimDistance = 0.0;
        aimTargetDeg = 0.0;
        aimTargetX = AIM_POINT_X_IN;
        aimTargetY = AIM_POINT_Y_LOW_IN;
        hasValidAim = false;
        lastValidAimTargetDeg = 0.0;
        turretReachable = true;

        telemetry.addLine("已初始化：底盘 + IMU云台 + Pinpoint双瞄点(60,60)/(60,84)");
        telemetry.addLine("X 自瞄开关；dpad 微调");
        telemetry.addLine("云台超出 ±150° 行程时手柄震动");
        telemetry.addLine(pinpointReady ? "Pinpoint: 已启用" : "Pinpoint: 未找到");
        telemetry.update();
    }

    @Override
    public void start() {
        prevStart = true;
        prevX = false;
    }

    @Override
    public void loop() {

        // ================= Pinpoint 更新 =================
        String pinpointStatusStr = "N/A";
        if (pinpointReady) {
            pinpoint.update();

            GoBildaPinpointDriver.DeviceStatus status = pinpoint.getDeviceStatus();
            pinpointStatusStr = status.toString();

            if (needSetStartPos && status == GoBildaPinpointDriver.DeviceStatus.READY) {
                pinpoint.setPosition(new Pose2D(
                        DistanceUnit.INCH,
                        START_X_IN, START_Y_IN,
                        AngleUnit.DEGREES, START_HEADING_DEG));
                needSetStartPos = false;
                pinX = START_X_IN;
                pinY = START_Y_IN;
                pinHeadingDeg = START_HEADING_DEG;
            }

            Pose2D pose = pinpoint.getPosition();
            pinX = pose.getX(DistanceUnit.INCH);
            pinY = pose.getY(DistanceUnit.INCH);
            pinHeadingDeg = pinpoint.getHeading(AngleUnit.DEGREES);

            // ===== 双瞄点选择 + 滞回 =====
            if (aimTargetY == AIM_POINT_Y_LOW_IN) {
                if (pinY > AIM_SWITCH_Y_IN + AIM_SWITCH_HYST_IN) {
                    aimTargetY = AIM_POINT_Y_HIGH_IN;
                }
            } else {
                if (pinY < AIM_SWITCH_Y_IN - AIM_SWITCH_HYST_IN) {
                    aimTargetY = AIM_POINT_Y_LOW_IN;
                }
            }
        }

        // ================= 底盘 Field-Centric =================
        double drive  = deadzone(-gamepad1.left_stick_y);
        double strafe = deadzone( gamepad1.left_stick_x);
        double turn   = deadzone( gamepad1.right_stick_x);

        double yawRad = imu.getRobotYawPitchRollAngles().getYaw(AngleUnit.RADIANS);
        double yawDeg = Math.toDegrees(yawRad);

        double cos = Math.cos(yawRad);
        double sin = Math.sin(yawRad);

        double strafeRobot =  strafe * cos + drive * sin;
        double driveRobot  = -strafe * sin + drive * cos;

        double fl = driveRobot + strafeRobot + turn;
        double fr = driveRobot - strafeRobot - turn;
        double bl = driveRobot - strafeRobot + turn;
        double br = driveRobot + strafeRobot - turn;

        double max = Math.max(Math.max(Math.abs(fl), Math.abs(fr)),
                              Math.max(Math.abs(bl), Math.abs(br)));
        if (max > 1.0) {
            fl /= max; fr /= max; bl /= max; br /= max;
        }

        frontLeft.setPower(fl);
        backLeft.setPower(bl);
        frontRight.setPower(fr);
        backRight.setPower(br);

        // ================= X：自瞄开关 =================
        boolean x = gamepad1.x;
        if (x && !prevX) {
            if (pinpointReady) {
                aimMode = !aimMode;
                hasValidAim = false;
                lastValidAimTargetDeg = 0.0;
            }
        }
        prevX = x;

        // ================= 自瞄目标角计算 =================
        if (aimMode && pinpointReady) {
            double dx = aimTargetX - pinX;
            double dy = aimTargetY - pinY;
            aimDistance = Math.hypot(dx, dy);

            if (aimDistance > AIM_MIN_DIST_IN) {
                aimTargetDeg = Math.toDegrees(Math.atan2(dy, dx));
                lastValidAimTargetDeg = aimTargetDeg;
                hasValidAim = true;
                turretTargetYawDeg = normalizeDeg(aimTargetDeg + aimOffsetDeg);
            } else {
                if (hasValidAim) {
                    aimTargetDeg = lastValidAimTargetDeg;
                    turretTargetYawDeg = normalizeDeg(aimTargetDeg + aimOffsetDeg);
                } else {
                    double currentTurretAbsDeg = normalizeDeg(
                            yawDeg
                            + TURRET_MOUNT_OFFSET_DEG
                            + TURRET_SIGN * (turretPos - TURRET_INIT)
                              * TURRET_SERVO_RANGE_DEG / TURRET_MECH_RATIO);
                    turretTargetYawDeg = currentTurretAbsDeg;
                    aimTargetDeg = normalizeDeg(currentTurretAbsDeg - aimOffsetDeg);
                }
            }
        } else if (pinpointReady) {
            aimDistance = Math.hypot(aimTargetX - pinX, aimTargetY - pinY);
        }

        // ================= dpad 微调 =================
        boolean left  = gamepad1.dpad_left;
        boolean right = gamepad1.dpad_right;

        if (left && !prevLeft) {
            if (aimMode) {
                aimOffsetDeg = normalizeDeg(aimOffsetDeg + TURRET_MANUAL_STEP_DEG);
            } else {
                turretTargetYawDeg = normalizeDeg(turretTargetYawDeg + TURRET_MANUAL_STEP_DEG);
            }
        }
        if (right && !prevRight) {
            if (aimMode) {
                aimOffsetDeg = normalizeDeg(aimOffsetDeg - TURRET_MANUAL_STEP_DEG);
            } else {
                turretTargetYawDeg = normalizeDeg(turretTargetYawDeg - TURRET_MANUAL_STEP_DEG);
            }
        }
        prevLeft = left;
        prevRight = right;

        // ================= IMU 云台绝对朝向锁定 =================
        double effectiveTargetDeg = normalizeDeg(turretTargetYawDeg + TURRET_MOUNT_OFFSET_DEG);
        double targetRad = Math.toRadians(effectiveTargetDeg);
        double yawErrorRad = normalizeRad(targetRad - yawRad);
        double yawErrorDeg = Math.toDegrees(yawErrorRad);

        if (Math.abs(yawErrorDeg) < TURRET_DEADBAND_DEG) {
            yawErrorDeg = 0.0;
        }

        double turretRelativeDeg = TURRET_SIGN * yawErrorDeg / TURRET_MECH_RATIO;
        double turretPosRaw = TURRET_INIT + turretRelativeDeg / TURRET_SERVO_RANGE_DEG;

        // 云台是否可达：原始位置在 [0,1] 内才可达
        turretReachable = (turretPosRaw >= TURRET_MIN && turretPosRaw <= TURRET_MAX);

        turretPos = clamp(turretPosRaw, TURRET_MIN, TURRET_MAX);

        turretLeft.setPosition(turretPos);
        turretRight.setPosition(turretPos);

        // 云台实际物理绝对朝向（含 180° 安装偏移）
        double turretActualAbsDeg = normalizeDeg(
                yawDeg
                + TURRET_MOUNT_OFFSET_DEG
                + TURRET_SIGN * (turretPos - TURRET_INIT)
                  * TURRET_SERVO_RANGE_DEG / TURRET_MECH_RATIO);

        // ================= 云台不可达时手柄震动 =================
            // ================= 云台不可达时手柄震动 =================
        if (aimMode && !turretReachable) {
            // rumble1: 左马达强度, rumble2: 右马达强度, durationMs: 持续时间(毫秒)
            // 每帧调用一次 → 只要云台超限就持续震动
            gamepad1.rumble(0.8, 0.8, 200); 
        } else {
            gamepad1.stopRumble(); // 震动停止
        }
        // ================= Intake + Transit =================
        double seriesPower = 0.0;
        double rt = gamepad1.right_trigger;
        double lt = gamepad1.left_trigger;
        if (rt > TRIGGER_THRESHOLD) seriesPower = rt;
        else if (lt > TRIGGER_THRESHOLD) seriesPower = -lt;

        intakeMotor.setPower(seriesPower);
        transitMotor.setPower(seriesPower);

        // ================= START =================
        boolean startBtn = gamepad1.start;
        if (startBtn && !prevStart) {
            imu.resetYaw();
            turretTargetYawDeg = START_HEADING_DEG;
            aimOffsetDeg = 0.0;
            hasValidAim = false;
            lastValidAimTargetDeg = 0.0;
            if (pinpointReady) {
                pinpoint.setPosition(new Pose2D(
                        DistanceUnit.INCH,
                        START_X_IN, START_Y_IN,
                        AngleUnit.DEGREES, START_HEADING_DEG));
                needSetStartPos = false;
            }
        }
        prevStart = startBtn;

        // ================= 遥测 =================
        telemetry.addData("IMU Yaw (°)", "%.1f", yawDeg);
        telemetry.addData("底盘 FL/BL/FR/BR", "%.2f / %.2f / %.2f / %.2f",
                fl, bl, fr, br);

        telemetry.addData("Pinpoint 状态", pinpointReady ? pinpointStatusStr : "未配置");
        if (pinpointReady) {
            telemetry.addData("Pinpoint X/Y (in)", "(%.2f, %.2f)", pinX, pinY);
            telemetry.addData("Pinpoint Heading (°)", "%.1f", pinHeadingDeg);
            telemetry.addData("Pinpoint - IMU 差值 (°)", "%.1f",
                    normalizeDeg(pinHeadingDeg - yawDeg));
            telemetry.addData("设置起点", needSetStartPos ? "等待 READY…" : "完成");
        }

        telemetry.addData("自瞄模式", aimMode ? "开 (X 关闭)" : "关 (X 开启)");
        telemetry.addData("当前瞄点 (in)", "(%.1f, %.1f)", aimTargetX, aimTargetY);
        telemetry.addData("切换阈值 y / 滞回", "%.1f / %.1f",
                AIM_SWITCH_Y_IN, AIM_SWITCH_HYST_IN);
        telemetry.addData("自瞄距离 (in)", "%.2f", aimDistance);
        telemetry.addData("指向瞄点绝对角 (°)", "%.1f", aimTargetDeg);
        telemetry.addData("自瞄偏移角 (°)", "%.1f", aimOffsetDeg);
        telemetry.addData("有效瞄准已建立?", hasValidAim ? "是" : "否");

        telemetry.addData("云台目标绝对角 (°)", "%.1f", turretTargetYawDeg);
        telemetry.addData("送给舵机目标角 (°)", "%.1f", effectiveTargetDeg);
        telemetry.addData("云台实际绝对角 (°)", "%.1f", turretActualAbsDeg);
        telemetry.addData("云台锁定误差 (°)", "%.1f", yawErrorDeg);
        telemetry.addData("云台舵机位置", "%.3f", turretPos);
        telemetry.addData("云台原始位置(未限幅)", "%.3f", turretPosRaw);
        telemetry.addData("云台可达?", turretReachable ? "是" : "否（顶限位，震动）");

        telemetry.addData("串联功率", "%.2f", seriesPower);

        if (aimMode && !turretReachable) {
            telemetry.addLine("※ 云台超出 ±150° 行程，已震动提醒");
        }
        if (aimMode && !hasValidAim) {
            telemetry.addLine("※ 距离瞄点太近，方向未定义；请把车移开 ≥6 英寸");
        }

        telemetry.update();
    }

    private static double clamp(double v, double lo, double hi) {
        return Math.max(lo, Math.min(hi, v));
    }
    private static double deadzone(double v) {
        return Math.abs(v) < STICK_DEADZONE ? 0.0 : v;
    }
    private static double normalizeRad(double a) {
        while (a >  Math.PI) a -= 2.0 * Math.PI;
        while (a < -Math.PI) a += 2.0 * Math.PI;
        return a;
    }
    private static double normalizeDeg(double a) {
        while (a >  180.0) a -= 360.0;
        while (a < -180.0) a += 360.0;
        return a;
    }
}
